// TP2 - Lecture LDR (R1) via pont avec R2 - GPIO22 (schema KiCad)
const int PIN_LDR = 22 ;
void setup() { : Cette fonction nets executer une seule fois au demarage
  Serial.begin(115200); : Demart la comunication
}
void loop() { : Repetition de la partie
  int brut = analogRead(PIN_LDR); // 0 .. 4095 : lit la valeure analogique du capteur LDR
  float tension = (brut / 4095.0) * 3.3; // estimation en volts : Transforme la valeur ADC en une tension
  Serial.print("ADC="); : Affiche une information sur le port serie
  Serial.print(brut);
  Serial.print(" U22=");
  Serial.print(tension, 2);
  Serial.println(" V");
  delay(300); :Attemps 300ms et puis recommence
}
